2. `WITH g AS (SELECT stm_ms - LAG(stm_ms) OVER (ORDER BY row_no) d FROM sample_flags WHERE session_id=3 AND bad_ms=0) SELECT SUM(d<0) FROM g`
3. `SELECT COUNT(*) FROM sample_flags WHERE session_id=3 AND bad_ms+bad_inv+bad_bms+bad_speed+bad_value>0`
4. `WITH r AS (SELECT bms_current*1.0/inv_dc_current x, ROW_NUMBER() OVER (ORDER BY bms_current*1.0/inv_dc_current) k, COUNT(*) OVER () n FROM sample_flags WHERE session_id=3 AND inv_dc_current>25 AND bad_inv=0 AND bad_bms=0) SELECT AVG(x), MAX(n) FROM r WHERE k IN ((n+1)/2, (n+2)/2)`
5. `WITH t AS (SELECT row_no, stm_ms, CAST(substr(ts_utc,12,2) AS REAL)*3600+CAST(substr(ts_utc,15,2) AS REAL)*60+CAST(substr(ts_utc,18,6) AS REAL) sec FROM sample WHERE session_id=3), s AS (SELECT row_no, stm_ms - LAG(stm_ms) OVER (ORDER BY row_no) dm, sec - LAG(sec) OVER (ORDER BY row_no) dt FROM t) SELECT COUNT(*), MIN(row_no), MAX(row_no), SUM(dt) FROM s WHERE dm<0 AND dt>60`
6. `SELECT COUNT(*), MIN(row_no), MAX(row_no), MAX(bms_current)-MIN(bms_current) FROM sample WHERE session_id=3 AND row_no BETWEEN 13819 AND 13872 AND inv_dc_current=85.6`

8. `WITH s AS (SELECT row_no, steering_angle a, LAG(steering_angle) OVER (ORDER BY row_no) p, LEAD(steering_angle) OVER (ORDER BY row_no) nx FROM sample WHERE session_id=3) SELECT COUNT(*), MIN(row_no), MAX(row_no) FROM s WHERE ABS(a-p)>30 AND ABS(a-nx)>30 AND row_no NOT IN (SELECT row_no FROM value_flag WHERE session_id=3 AND column_name='steering_angle')`
9. `WITH s AS (SELECT row_no, rl a, LAG(rl) OVER (ORDER BY row_no) p, LEAD(rl) OVER (ORDER BY row_no) nx FROM sample WHERE session_id=3) SELECT COUNT(*), MIN(row_no), MAX(row_no) FROM s WHERE ABS(a-p)>5 AND ABS(a-nx)>5 AND ABS(p-nx)<2`
